add_executable(gdi-frame-diff-test frame-diff-test.c)
target_include_directories(gdi-frame-diff-test PRIVATE ../include)
add_executable(gdi-exclusions-compare-test exclusions-compare-test.c)
set(WGDI_TEST_TARGETS gdi-frame-diff-test gdi-exclusions-compare-test)
if(APPLE)
    find_package(Threads REQUIRED)
    add_executable(gdi-content-cache-test content-cache-test.cpp)
    target_include_directories(gdi-content-cache-test PRIVATE ../src)
    target_link_libraries(gdi-content-cache-test PRIVATE Threads::Threads)
    add_executable(gdi-coreml-client-test coreml-client-test.cpp ../src/coreml-client.cpp)
    target_include_directories(gdi-coreml-client-test PRIVATE ../include ../src)
    list(APPEND WGDI_TEST_TARGETS gdi-content-cache-test gdi-coreml-client-test)
endif()
if(WGDI_TEST_SANITIZERS)
    if(CMAKE_C_COMPILER_ID MATCHES "Clang|GNU" AND CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        set(WGDI_TEST_SANITIZER_FLAGS -fsanitize=address,undefined -fno-sanitize-recover=all
            -fno-omit-frame-pointer)
    else()
        message(FATAL_ERROR "WGDI_TEST_SANITIZERS requires Clang or GNU compilers")
    endif()
endif()
foreach(TARGET IN LISTS WGDI_TEST_TARGETS)
    target_compile_options(${TARGET} PRIVATE -UNDEBUG -Wall -Wextra)
    if(WGDI_TEST_SANITIZERS)
        target_compile_options(${TARGET} PRIVATE ${WGDI_TEST_SANITIZER_FLAGS})
        target_link_options(${TARGET} PRIVATE ${WGDI_TEST_SANITIZER_FLAGS})
    endif()
endforeach()
add_test(NAME gdi-frame-difference-and-stability COMMAND gdi-frame-diff-test)
set_tests_properties(gdi-frame-difference-and-stability PROPERTIES TIMEOUT 15 LABELS "test-only;headless")
add_test(NAME gdi-exclusion-masked-comparison COMMAND gdi-exclusions-compare-test)
set_tests_properties(gdi-exclusion-masked-comparison PROPERTIES TIMEOUT 15 LABELS "test-only;headless")
if(APPLE)
    add_test(NAME gdi-content-addressed-store COMMAND gdi-content-cache-test)
    set_tests_properties(gdi-content-addressed-store PROPERTIES TIMEOUT 60 LABELS "test-only;headless")
    add_test(NAME gdi-coreml-client-failure-fallback COMMAND gdi-coreml-client-test)
    set_tests_properties(gdi-coreml-client-failure-fallback PROPERTIES TIMEOUT 45 LABELS "test-only;headless")
endif()
